cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico2_w CACHE STRING "" FORCE)
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "" FORCE)
set(SWITCH_PICO_INPUT_BACKEND UART CACHE STRING "" FORCE)
set(SWITCH_PICO_UART_CONTROLLERS 1 CACHE STRING "" FORCE)
set(SWITCH_PICO_SYS_CLOCK_MHZ 150 CACHE STRING "" FORCE)
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(seedsow_standalone C CXX ASM)
add_subdirectory(${SWITCH_PICO_SOURCE} upstream)
get_target_property(sources switch-pico SOURCES)
list(FILTER sources EXCLUDE REGEX "/main\\.cpp$")
set_property(TARGET switch-pico PROPERTY SOURCES "${sources}")
target_sources(switch-pico PRIVATE main.cpp engine.c
 ${SWITCH_PICO_SOURCE}/src/firmware/platform/pico/bootsel_pairing_button.cpp)
target_link_libraries(switch-pico PRIVATE pico_flash)
get_target_property(definitions switch-pico COMPILE_DEFINITIONS)
list(FILTER definitions EXCLUDE REGEX "^PICO_PROGRAM_NAME=")
set_property(TARGET switch-pico PROPERTY COMPILE_DEFINITIONS "${definitions}")
pico_set_program_name(switch-pico "Seedsow Pico 2 W standalone")
